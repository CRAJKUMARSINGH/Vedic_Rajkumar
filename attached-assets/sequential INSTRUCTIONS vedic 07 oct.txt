```typescript
// supabase/functions/calculate-kundli/index.ts
import { serve } from 'https://deno.land/std@0.168.0/http/server.ts';
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders });
  }

  try {
    const { birthDate, birthTime, latitude, longitude, ayanamsa = 'Lahiri' } = await req.json();
    
    // Validate inputs
    if (!birthDate || !birthTime) {
      return new Response(JSON.stringify({ error: 'Missing required fields' }), {
        status: 400,
        headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      });
    }

    // Call Swiss Ephemeris WASM on edge
    const chart = await calculateSiderealChart({
      birthDate,
      birthTime,
      latitude,
      longitude,
      ayanamsa,
    });

    // Cache response in Redis/Upstash (optional)
    return new Response(JSON.stringify(chart), {
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
      status: 200,
    });
  } catch (error) {
    return new Response(JSON.stringify({ error: error.message }), {
      status: 500,
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
    });
  }
});
```

```typescript
// src/api/client.ts
import axios, { AxiosInstance, AxiosError } from 'axios';

const API_BASE = import.meta.env.VITE_API_URL || 'https://your-project.supabase.co/functions/v1';

class APIClient {
  private client: AxiosInstance;
  
  constructor() {
    this.client = axios.create({
      baseURL: API_BASE,
      timeout: 30000,
      headers: {
        'Content-Type': 'application/json',
      },
    });
    
    // Request interceptor for auth
    this.client.interceptors.request.use((config) => {
      const token = localStorage.getItem('access_token');
      if (token) {
        config.headers.Authorization = `Bearer ${token}`;
      }
      return config;
    });
    
    // Response interceptor for caching
    this.client.interceptors.response.use(
      (response) => {
        // Cache GET requests
        if (response.config.method === 'get') {
          this.cache.set(response.config.url!, response.data);
        }
        return response;
      },
      (error: AxiosError) => {
        if (error.response?.status === 429) {
          // Rate limit - fallback to client-side
          return this.fallbackToClient(error.config);
        }
        return Promise.reject(error);
      }
    );
  }
  
  async calculateKundli(data: KundliInput): Promise<KundliChart> {
    const response = await this.client.post('/calculate-kundli', data);
    return response.data;
  }
}

export const api = new APIClient();
```

### STEP 15: Create API Client

**Create file:** `src/api/client.ts`

```typescript
import { KundliInput, KundliChart, Panchang, MatchmakingResult, TransitSnapshot } from './types';

const API_BASE = (import.meta.env.VITE_SUPABASE_URL || '') + '/functions/v1';

class APIError extends Error {
  constructor(public status: number, message: string) {
    super(message);
  }
}

class VedicAPI {
  private cache = new Map<string, unknown>();

  private async fetchWithAuth(url: string, options?: RequestInit): Promise<Response> {
    const token = localStorage.getItem('supabase_access_token');
    const headers: Record<string, string> = {
      'Content-Type': 'application/json',
      ...(token && { Authorization: `Bearer ${token}` }),
      ...((options?.headers as Record<string, string>) || {}),
    };

    const response = await fetch(url, {
      ...options,
      headers,
    });

    if (!response.ok) {
      const error = await response.json().catch(() => ({ message: 'Unknown error' }));
      throw new APIError(response.status, error.error || error.message);
    }
    return response;
  }

  async calculateKundli(input: KundliInput): Promise<KundliChart> {
    const cacheKey = `kundli_${JSON.stringify(input)}`;
    if (this.cache.has(cacheKey)) return this.cache.get(cacheKey) as KundliChart;

    const response = await this.fetchWithAuth(`${API_BASE}/calculate-kundli`, {
      method: 'POST',
      body: JSON.stringify(input),
    });
    const result = await response.json();
    this.cache.set(cacheKey, result);
    return result;
  }

  async getPanchang(date: string, location: { latitude: number; longitude: number }): Promise<Panchang> {
    const response = await this.fetchWithAuth(`${API_BASE}/get-panchang`, {
      method: 'POST',
      body: JSON.stringify({ date, ...location }),
    });
    return response.json();
  }

  async calculateMatchmaking(maleChart: KundliChart, femaleChart: KundliChart): Promise<MatchmakingResult> {
    const response = await this.fetchWithAuth(`${API_BASE}/calculate-matchmaking`, {
      method: 'POST',
      body: JSON.stringify({ maleChart, femaleChart }),
    });
    return response.json();
  }

  async calculateTransits(natalChart: KundliChart, date: string): Promise<TransitSnapshot> {
    const response = await this.fetchWithAuth(`${API_BASE}/calculate-transits`, {
      method: 'POST',
      body: JSON.stringify({ natalChart, date }),
    });
    return response.json();
  }

  clearCache(): void {
    this.cache.clear();
  }
}

export const vedicApi = new VedicAPI();
export { APIError };
```

```typescript
import { useQuery, useMutation, UseQueryOptions } from '@tanstack/react-query';
import { vedicApi } from './client';
import { KundliInput, KundliChart, Panchang, MatchmakingResult, TransitSnapshot } from './types';

export const useKundli = (
  input: KundliInput | null,
  options?: Omit<UseQueryOptions<KundliChart>, 'queryKey' | 'queryFn'>
) => {
  return useQuery({
    queryKey: ['kundli', input],
    queryFn: () => vedicApi.calculateKundli(input!),
    enabled: !!input,
    staleTime: 1000 * 60 * 60 * 24, // 24 hours
    ...options,
  });
};

export const usePanchang = (date: string, location: { latitude: number; longitude: number }) => {
  return useQuery({
    queryKey: ['panchang', date, location],
    queryFn: () => vedicApi.getPanchang(date, location),
    enabled: !!date && !!location,
    staleTime: 1000 * 60 * 60, // 1 hour
  });
};

export const useMatchmaking = () => {
  return useMutation({
    mutationFn: ({ maleChart, femaleChart }: {
      maleChart: KundliChart;
      femaleChart: KundliChart;
    }) => vedicApi.calculateMatchmaking(maleChart, femaleChart),
  });
};

export const useTransits = (natalChart: KundliChart | null, date: string) => {
  return useQuery({
    queryKey: ['transits', natalChart, date],
    queryFn: () => vedicApi.calculateTransits(natalChart!, date),
    enabled: !!natalChart && !!date,
  });
};
```

### STEP 6: Create Flagsmith Provider

**File:** `src/features/flags/provider.tsx`

```typescript
import React, { createContext, useContext, useEffect, useState, ReactNode } from 'react';
import flagsmith from 'flagsmith';
import { UserTier } from '../tier/config';

interface FlagsmithContextType {
  isReady: boolean;
  hasFeature: (flag: string) => boolean;
  getValue: <T>(flag: string, defaultValue: T) => T;
  identify: (userId: string, traits: Record<string, any>) => Promise<void>;
  logout: () => Promise<void>;
}

const FlagsmithContext = createContext<FlagsmithContextType | null>(null);

export const FlagsmithProvider: React.FC<{ children: ReactNode }> = ({ children }) => {
  const [isReady, setIsReady] = useState(false);

  useEffect(() => {
    flagsmith.init({
      environmentID: import.meta.env.VITE_FLAGSMITH_ENV_ID,
      cacheFlags: true,
      defaultFlags: {
        kundli: { default: true },
        prashna: { default: true },
        matchmaking: { default: true },
        panchang: { default: true },
        dasha_timeline: { default: false },
        transit_alerts: { default: false },
      },
    }).then(() => {
      setIsReady(true);
    });
  }, []);

  const hasFeature = (flag: string): boolean => {
    return flagsmith.hasFeature(flag);
  };

  const getValue = <T,>(flag: string, defaultValue: T): T => {
    return flagsmith.getValue(flag, defaultValue);
  };

  const identify = async (userId: string, traits: Record<string, any>) => {
    await flagsmith.identify(userId, traits);
  };

  const logout = async () => {
    await flagsmith.logout();
  };

  return (
    <FlagsmithContext.Provider value={{ isReady, hasFeature, getValue, identify, logout }}>
      {children}
    </FlagsmithContext.Provider>
  );
};

export const useFlagsmith = () => {
  const context = useContext(FlagsmithContext);
  if (!context) throw new Error('useFlagsmith must be used within FlagsmithProvider');
  return context;
};
```

### STEP 7: Create useFeature Hook

**File:** `src/features/flags/useFeature.ts`

```typescript
import { useFlagsmith } from './provider';
import { TIER_FEATURES, UserTier, TIER_LIMITS } from '../tier/config';
import { FEATURE_FLAGS, FeatureFlag } from './config';

interface UseFeatureResult {
  enabled: boolean;
  isLoading: boolean;
  reason: 'flag' | 'tier' | 'limit' | 'ready';
}

export const useFeature = (
  flag: FeatureFlag,
  userTier: UserTier = 'anonymous'
): UseFeatureResult => {
  const { isReady, hasFeature } = useFlagsmith();

  // Check tier first
  const tierFeatures = TIER_FEATURES[userTier];
  const hasTierAccess = tierFeatures.includes(flag);

  // Check feature flag
  const flagEnabled = hasFeature(flag);

  // Determine final state
  if (!isReady) {
    return { enabled: false, isLoading: true, reason: 'ready' };
  }

  if (!hasTierAccess) {
    return { enabled: false, isLoading: false, reason: 'tier' };
  }

  if (!flagEnabled) {
    return { enabled: false, isLoading: false, reason: 'flag' };
  }

  return { enabled: true, isLoading: false, reason: 'flag' };
};

export const useTierLimits = (userTier: UserTier) => {
  return TIER_LIMITS[userTier];
};

export const useCanAccessFeature = (
  flag: FeatureFlag,
  userTier: UserTier,
  currentUsage: number,
  limitKey: keyof typeof TIER_LIMITS[UserTier]
): boolean => {
  const limits = TIER_LIMITS[userTier];
  const limit = limits[limitKey] as number;

  // Unlimited (-1) or within limit
  const withinLimit = limit === -1 || currentUsage < limit;

  const { enabled } = useFeature(flag, userTier);

  return enabled && withinLimit;
};
```

### STEP 9: Create A/B Testing Hook

**File:** `src/features/ab/useExperiment.ts`

```typescript
import { useFlagsmith } from '../flags/provider';

type ExperimentVariant = 'control' | 'treatment' | 'treatment-a' | 'treatment-b';

interface ExperimentResult<T> {
  variant: ExperimentVariant;
  value: T;
  track: (event: string, metadata?: Record<string, any>) => void;
}

export const useExperiment = <T = string>(
  experimentKey: string,
  variants: Record<ExperimentVariant, T>
): ExperimentResult<T> => {
  const { getValue } = useFlagsmith();

  // Get assigned variant from Flagsmith
  const assignedVariant = getValue<ExperimentVariant>(experimentKey, 'control');

  // Get value for variant
  const value = variants[assignedVariant] ?? variants.control;

  // Track experiment event
  const track = (event: string, metadata?: Record<string, any>) => {
    if (typeof window !== 'undefined' && (window as any).gtag) {
      (window as any).gtag('event', event, {
        experiment_id: experimentKey,
        variant: assignedVariant,
        ...metadata,
      });
    }
  };

  return {
    variant: assignedVariant,
    value,
    track,
  };
};
```

```sql
-- Add org_id to charts if not exists
ALTER TABLE charts ADD COLUMN IF NOT EXISTS org_id UUID REFERENCES organizations(id);
```

### STEP 8: Create Permission Hook

**File:** `src/auth/permissions/usePermission.ts`

```typescript
import { useUser } from '@clerk/clerk-react';
import { useQuery } from '@tanstack/react-query';
import { supabase } from '@/integrations/supabase/client';
import { Permission, UserRole, ROLE_DEFINITIONS } from '../roles/config';

interface UserOrgRole {
  org_id: string;
  role: UserRole;
  permissions: Permission[];
}

export const useUserRole = (orgId?: string) => {
  const { user } = useUser();

  return useQuery({
    queryKey: ['userRole', user?.id, orgId],
    queryFn: async (): Promise<UserOrgRole | null> => {
      if (!user) return null;

      const query = supabase
        .from('user_roles')
        .select('org_id, role, permissions')
        .eq('user_id', user.id);

      if (orgId) {
        query.eq('org_id', orgId);
      }

      const { data, error } = await query.single();
      if (error) return null;
      return data;
    },
    enabled: !!user,
  });
};

export const useHasPermission = (
  permission: Permission,
  orgId?: string
): boolean => {
  const { data: userRole } = useUserRole(orgId);

  if (!userRole) return false;

  const roleDef = ROLE_DEFINITIONS[userRole.role];

  // Check wildcard permissions
  if (roleDef.permissions.includes('*')) return true;
  if (roleDef.permissions.includes(permission)) return true;

  // Check wildcard for resource type
  const resourceType = permission.split(':')[0];
  if (roleDef.permissions.includes(`${resourceType}:*`)) return true;

  return false;
};

export const useIsAdmin = (orgId?: string): boolean => {
  const { data: userRole } = useUserRole(orgId);
  if (!userRole) return false;
  return ROLE_DEFINITIONS[userRole.role].isAdmin;
};
```

```typescript
import React, { ReactNode } from 'react';
import { useUserRole } from '@/auth/permissions/usePermission';
import { UserRole } from '@/auth/roles/config';
import { Shield, Loader2 } from 'lucide-react';

interface RoleGateProps {
  allowedRoles: UserRole[];
  orgId?: string;
  children: ReactNode;
  fallback?: ReactNode;
}

export const RoleGate: React.FC<RoleGateProps> = ({
  allowedRoles,
  orgId,
  children,
  fallback,
}) => {
  const { data: userRole, isLoading } = useUserRole(orgId);

  if (isLoading) {
    return (
      <div className="flex items-center justify-center p-4">
        <Loader2 className="h-5 w-5 animate-spin" />
      </div>
    );
  }

  if (!userRole || !allowedRoles.includes(userRole.role)) {
    return fallback || (
      <div className="flex flex-col items-center justify-center p-6 border rounded-lg bg-muted">
        <Shield className="h-8 w-8 mb-2 text-muted-foreground" />
        <p className="text-sm text-muted-foreground">
          You don't have permission to access this feature.
        </p>
      </div>
    );
  }

  return <>{children}</>;
};
```

```typescript
import React, { useState } from 'react';
import { useUser } from '@clerk/clerk-react';
import { Button } from '@/components/ui/button';
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card';

        {isMFAEnabled ? (
          <div className="flex items-center gap-2 text-green-600">
            <Check className="h-5 w-5" />
            <span>MFA is enabled</span>
          </div>
        ) : (
          <div className="space-y-4">
            <div className="flex items-center gap-4">
              <Smartphone className="h-8 w-8 text-muted-foreground" />
              <div>
```
